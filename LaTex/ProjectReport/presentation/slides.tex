\begin{frame}{Real-World Need}
  \head{The hospital floor is a queueing system whose tasks aren't interchangeable.}
  \begin{itemize}
    \item Patients differ in \emph{acuity} (CRIT/HIGH/MED/LOW) and
          \emph{disease}; both shape the workup.
    \item Tests have hard \emph{precedence} (Troponin $\to$ ECHO $\to$ Stress).
    \item Workers are \emph{role-typed} (PGY1, INTERN, NURSE) and
          shift-bounded.
    \item Shift exceeded $\Rightarrow$ overtime; overtime exceeded
          $\Rightarrow$ burnout.
  \end{itemize}
  \vspace{0.6em}
  \begin{insight}
    \head{What an algorithm-driven scheduler must do.}\;
    Respect roles \;|\; respect precedence \;|\; minimise urgent-patient
    wait \;|\; honour finite shifts \;|\; report feasibility loudly.
  \end{insight}
\end{frame}

% =============================================================================
%  Slide 4 --- Existing Gaps
% =============================================================================
\begin{frame}{Existing Gaps in Current Practice}
  \begin{columns}[T,onlytextwidth]
    \column{0.5\textwidth}
      \head{Static heuristics.}
      \begin{itemize}
        \item One scalar priority $\Rightarrow$ entire Pareto regions
              unreachable.
        \item No notion of role-aware matching.
        \item No way to compare two algorithm variants on the
              same input.
      \end{itemize}
    \column{0.5\textwidth}
      \head{No defensible accountability.}
      \begin{itemize}
        \item No feasibility verdict --- shifts silently overrun.
        \item No per-acuity timing report.
        \item No record of \emph{which trade-offs} were chosen and
              \emph{why}.
      \end{itemize}
  \end{columns}
  \vspace{0.4em}
  \begin{insight}
    \head{claim:}
    a scheduling engine that can be \emph{benchmarked} on multiple
    axes is more valuable than one that ships ``the optimal algorithm''.
  \end{insight}
\end{frame}

%  Slide 5 --- Project Goals

\begin{frame}{Project Goals}
  \begin{enumerate}
    \item \head{Correct, deterministic discrete-event scheduler}
          with two pluggable policy axes (ordering, matching).
    \item \head{Lexicographic priority abstraction} (\code{LexKey})
          --- a hand-curated 7-policy catalogue subsuming the
          legacy scalar ordering.
    \item \head{Multi-objective evaluation}: Pareto frontier on
          (urgent-care, makespan, max-OT) + NSGA-II crowding.
    \item \head{Six-mode benchmark suite} that isolates one
          algorithmic axis at a time.
  \end{enumerate}
\end{frame}

\begin{frame}[fragile]{Architecture --- The Engine, One Tick at a Time}
  \begin{columns}[T,onlytextwidth]
    \column{0.42\textwidth}
      \head{Six things every tick.}
      \begin{enumerate}\small
        \item Look at unfinished tasks.
        \item Check rules: which are ready?
        \item Sort the ready set (ordering).
        \item Find a free, role-matched worker.
        \item Assign; move time forward.
        \item Unblock dependents; loop.
      \end{enumerate}
      \vspace{0.4em}
      \begin{center}\footnotesize\color{slate}
        Two pluggable axes:\\
        \head{ordering} (what next) \;|\;
        \head{matching} (who does it).
      \end{center}
    \column{0.55\textwidth}
      \begin{tcolorbox}[enhanced, breakable=false,
                        colback=boxbg, colframe=ruleblue,
                        boxrule=0.4pt, arc=2pt,
                        left=4pt, right=4pt, top=0pt, bottom=0pt]
\begin{Verbatim}[fontsize=\tiny]
   ready tasks (LexKey)
            ▼
   ┌────────────────────┐
   │ OrderingStrategy   │  ◀── what next?
   │  FCFS / Priority   │
   └─────────┬──────────┘
             ▼ drain_up_to(|W|)
        ╔════════════╗
        ║ run_engine ║
        ╚══════╤═════╝
               ▼ batch
   ┌────────────────────┐
   │ MatchingStrategy   │  ◀── who does it?
   │ Greedy / Fair / JV │
   └─────────┬──────────┘
             ▼
   Schedule · Capacity · Metrics
\end{Verbatim}

\begin{frame}{Algorithms (1/2) --- Ordering and Lex Policy}
  \head{Ordering: \code{FCFS} vs.\ \code{Priority}.}
  \begin{itemize}
    \item FCFS = sorted-by-id deque (deterministic FIFO).
    \item Priority = min-heap on \code{LexKey}.
  \end{itemize}
  \vspace{0.3em}
  \head{LexKey = lexicographic priority key (smaller = more urgent).}
  \begin{center}
    \footnotesize
    \begin{tabular}{l l}
      \toprule
      \head{Policy} & \head{Key shape} \\
      \midrule
      \code{acuity\_first}        & \code{(acuity, task.priority, hash)} \\
      \code{priority\_first}      & \code{(task.priority, acuity, hash)} \\
      \code{acuity\_short\_first} & \code{(acuity, +duration, ..., hash)} \\
      \code{acuity\_long\_first}  & \code{(acuity, $-$duration, ..., hash)} \\
      \code{acuity\_old\_first}   & \code{(acuity, $-$age, ..., hash)} \\
      \code{acuity\_young\_first} & \code{(acuity, $+$age, ..., hash)} \\
      \code{duration\_first}      & \code{(+duration, ..., hash)} \\
      \bottomrule
    \end{tabular}
  \end{center}
  \vspace{0.4em}\par
  \begin{center}\footnotesize\color{slate}
    Why lex over scalar: a scalar weighted-sum cannot pick between two
    schedules that agree on the dominant axis. Lex keys can.
  \end{center}
\end{frame}

\begin{frame}{Algorithms (2/2) --- Matching}
  \head{Three matchers, one interface.}
  \begin{center}
    \footnotesize
    \begin{tabular}{l l}
      \toprule
      \head{Matcher} & \head{Rule (in plain English)} \\
      \midrule
      \code{Greedy}                  & Pick the role-matched worker who's free earliest. \\
      \code{FairGreedy(\(\alpha\))}  & Blend ``earliest free'' with ``most shift left''
                                       (pure water-fill). \\
      \code{Hungarian}               & Globally optimal 1-to-1 pairing per tick. \\
      \bottomrule
    \end{tabular}
  \end{center}
  \vspace{0.4em}
  \begin{insight}
    \head{Per-tick behaviour matters.}\;
    Greedy can stack many tasks on one worker in a single tick;
    Hungarian commits at most one task per worker per tick, so
    overflow tasks defer to the next tick. This single difference
    drives the empirical results we'll see next.
  \end{insight}
\end{frame}

  \begin{insight}
    \head{Hungarian is \emph{actively bad} for critical patients here.}\;
    1-to-1-per-tick deferral on identical-key CRIT bursts. Greedy
    stacks them in one tick. Clinical correctness, not just runtime.
  \end{insight}
\end{frame}

% =============================================================================
%  Slide 19 --- Key Insights
% =============================================================================
\begin{frame}{Key Insights}
  \begin{enumerate}
    \item \head{Ordering moves more than matching} on a roster
          past capacity. Tie-breaks dominate.
    \item \head{Globally-optimal-per-tick is the wrong invariant}
          when priority keys collide. Greedy stacks; Hungarian defers.
    \item \head{Fair-mode water-filling needs heterogeneous initial
          load to engage.} Mechanism correct; dataset doesn't exercise it.
    \item \head{No scheduling change creates capacity.} 20{,}210~min
          of overtime are invariant.
  \end{enumerate}
  \vspace{0.6em}
  \begin{insight}
    \head{Recommended production config.}\;
    \code{Priority} $+$ \code{Greedy} $+$ \code{acuity\_young\_first} LexPolicy.\\
    \head{Recommended staffing change.}\;
    Add nurses. The roster is at $\sim$5$\times$ overload.
  \end{insight}
\end{frame}

\begin{frame}{Conclusion}
  \head{What I implemented.}
  \begin{itemize}\small
    \item A discrete-event scheduler with two pluggable policy axes
          (ordering, matching).
    \item Three matchers (Greedy, FairGreedy, Hungarian-JV) and a
          seven-policy lexicographic priority catalogue.
    \item A multi-objective Pareto frontier with NSGA-II crowding
          selection.
    \item A six-mode benchmark suite over a 100-patient, 10-worker ward.
  \end{itemize}
  \vspace{0.3em}
  \head{What the experiments showed.}
  \begin{itemize}\small
    \item On this dataset, the ordering axis moves the urgent-care
          metric more than the matching axis does.
    \item Lex tie-breaks reach Pareto points that a scalar
          weighted-sum priority cannot.
    \item Hungarian's per-tick optimality regressed CRIT timing on
          this dataset --- a 1-to-1-per-tick contract issue, not a
          solver bug.
    \item Total overtime is invariant across every strategy --- a
          workforce question, not an algorithm question.
  \end{itemize}
\end{frame}
