\section{Command-line reference}
\label{app:cli}

\subsection{\code{cardio\_scheduler --help}}

\begin{lstlisting}[language=bash,basicstyle=\ttfamily\small,
                   numbers=none,frame=single,backgroundcolor=\color{boxbg}]
Usage: cardio_scheduler [options] [data_dir]

  data_dir                 directory of the four JSON files
                           (default: ./data, falls back to ./src/data)

  --bench-2x2              FCFS/Priority * Greedy/Hungarian matrix
  --bench-pareto           Pareto sweep over lex policies x seeds
  --bench-fair             throughput vs worker-load stdev
  --bench-adversarial      synthetic wards built to break each algo
  --bench-urgency          per-acuity timing across 5 matchers

  --overtime-threshold N   OT budget (min) before INFEASIBLE
  -h, --help               print this message
\end{lstlisting}

\subsection{Data-directory resolution}

\begin{itemize}
  \item Probes \code{./data} first; falls back to
        \code{./src/data}.
  \item Picks the first that contains \code{diseases.json}.
  \item On failure: exits with code 2 and an informative error.
\end{itemize}
