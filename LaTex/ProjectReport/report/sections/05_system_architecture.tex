\section{System Architecture}
\label{sec:architecture}

\begin{itemize}
  \item One C++ executable (\code{cardio\_scheduler}) compiled
        from $\sim$3{,}024 LoC across 17 source files.
  \item Single header-only third-party dependency
        (\code{nlohmann::json}).
  \item Five-layer stack with narrow inter-layer interfaces.
\end{itemize}

\subsection{Five-layer stack}

\begin{itemize}
  \item \textbf{L1 --- Inputs.}
        Four JSON files: \code{diseases.json} (418),
        \code{tests.json} (183), \code{test\_dependencies.json} (325
        edges), \code{sample\_ward.json} (100 patients, 10 workers).
  \item \textbf{L2 --- Loader \& ward assembly.}
        Throws on missing files / unknown enum labels (fail-fast).
        \code{build\_ward} denormalises every patient $\times$ test
        pair into a unique task instance keyed
        \code{<test\_id>\_<patient\_id>}. Cross-patient edges
        forbidden. Reverse map \code{task\_to\_patient} populated.
  \item \textbf{L3 --- Engine core.}
        \code{run\_engine} (\code{allocator.cpp:79}) drives the loop;
        accepts \code{EngineConfig\{ordering, matching, policy,
        policy\_seed, label\}} and three optional out-pointers.
        Engine doesn't know the concrete strategy.
  \item \textbf{L4 --- Bench dispatch + Pareto tooling.}
        Each cell deep-copies \code{tasks}, \code{workers},
        \code{patients} (engine mutates in-place). Multi-objective
        tooling: \code{compute\_objectives},
        \code{pareto\_frontier}, \code{top\_k\_by\_crowding}.
  \item \textbf{L5 --- CLI + ward editor.}
        \code{main.cpp} dispatches to one of six bench modes
        (Appx.~\ref{app:cli}); \code{index.html} ($\sim$800 LoC,
        single-page, no runtime) edits \code{sample\_ward.json}.
\end{itemize}

\subsection{Module dependency}

\begin{itemize}
  \item Engine core depends only on the strategy interface
        (\code{strategies.h}) and the dependency graph
        (\code{graph.cpp}).
  \item Bench layer depends on the engine; the engine is unaware of
        the bench layer.
  \item Strategies (\code{strategies.cpp}, \code{hungarian.cpp},
        \code{policies.cpp}) depend only on \code{strategies.h} and
        the data models in \code{models/models.h}.
\end{itemize}
